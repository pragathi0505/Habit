import { useState, useEffect, useRef } from "react";

// ---- constants -------------------------------------------------------

const INK = "#2B2622";
const PAPER = "#F6F2EA";
const PAPER_RAISED = "#FBF8F2";
const RULE = "#D9CFBE";
const CLAY = "#B5502F";
const CLAY_SOFT = "#E3C3B3";
const MOSS = "#5C6B4E";
const MOSS_SOFT = "#D9E0CE";
const INK_FAINT = "#8A7F70";

const todayKey = () => {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
    d.getDate()
  ).padStart(2, "0")}`;
};

const prettyDate = (key) => {
  const d = new Date(key + "T00:00:00");
  return d.toLocaleDateString("en-IN", { weekday: "long", day: "numeric", month: "long" });
};

const yesterdayKey = (key) => {
  const d = new Date(key + "T00:00:00");
  d.setDate(d.getDate() - 1);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
    d.getDate()
  ).padStart(2, "0")}`;
};

const emptyEntry = () => ({
  recallDone: false,
  recallText: "",
  topic: "",
  freeWrite: "",
  gaps: "",
  oneIdea: "",
  oneApplication: "",
  oneAction: "",
  urgeLogged: false,
  urgeMove: "",
  urgeWorked: null, // true / false / null
  loggedAt: null,
});

// ---- tiny building blocks ---------------------------------------------

function SeedMark() {
  // signature element: a small hand-drawn "loop" glyph, standing for the
  // recall -> write -> apply cycle. Rendered once, quietly, near the masthead.
  return (
    <svg width="34" height="34" viewBox="0 0 34 34" fill="none">
      <path
        d="M8 20c0-6 5-11 11-11 4 0 7.5 2.3 9 5.8"
        stroke={CLAY}
        strokeWidth="2"
        strokeLinecap="round"
      />
      <path d="M23 9.5l5 5.6-6.8 1.4" stroke={CLAY} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
      <path
        d="M26 14c0 6-5 11-11 11-4 0-7.5-2.3-9-5.8"
        stroke={MOSS}
        strokeWidth="2"
        strokeLinecap="round"
      />
      <path d="M11 24.5L6 18.9l6.8-1.4" stroke={MOSS} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    </svg>
  );
}

function Field({ label, hint, value, onChange, rows = 3, placeholder }) {
  return (
    <div style={{ marginBottom: 18 }}>
      <div
        style={{
          display: "flex",
          justifyContent: "space-between",
          alignItems: "baseline",
          marginBottom: 6,
        }}
      >
        <label
          style={{
            fontFamily: "'Newsreader', Georgia, serif",
            fontStyle: "italic",
            fontSize: 15.5,
            color: INK,
          }}
        >
          {label}
        </label>
        {hint && (
          <span style={{ fontSize: 11, color: INK_FAINT, fontFamily: "'IBM Plex Mono', monospace" }}>
            {hint}
          </span>
        )}
      </div>
      <textarea
        value={value}
        onChange={(e) => onChange(e.target.value)}
        rows={rows}
        placeholder={placeholder}
        style={{
          width: "100%",
          boxSizing: "border-box",
          background: PAPER_RAISED,
          border: `1px solid ${RULE}`,
          borderRadius: 3,
          padding: "10px 12px",
          fontFamily: "'IBM Plex Mono', monospace",
          fontSize: 14,
          lineHeight: 1.55,
          color: INK,
          resize: "vertical",
          outline: "none",
        }}
        onFocus={(e) => (e.target.style.borderColor = CLAY)}
        onBlur={(e) => (e.target.style.borderColor = RULE)}
      />
    </div>
  );
}

function SectionLabel({ n, children }) {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 14 }}>
      <div
        style={{
          width: 22,
          height: 22,
          borderRadius: "50%",
          border: `1.5px solid ${CLAY}`,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          fontFamily: "'IBM Plex Mono', monospace",
          fontSize: 11,
          color: CLAY,
          flexShrink: 0,
        }}
      >
        {n}
      </div>
      <h3
        style={{
          fontFamily: "'Newsreader', Georgia, serif",
          fontWeight: 600,
          fontSize: 19,
          color: INK,
          margin: 0,
          letterSpacing: "-0.01em",
        }}
      >
        {children}
      </h3>
    </div>
  );
}

function Card({ children, style }) {
  return (
    <div
      style={{
        background: PAPER,
        border: `1px solid ${RULE}`,
        borderRadius: 4,
        padding: "20px 18px",
        marginBottom: 16,
        ...style,
      }}
    >
      {children}
    </div>
  );
}

// ---- main app -----------------------------------------------------------

export default function RecallLoop() {
  const [ready, setReady] = useState(false);
  const [dateKey] = useState(todayKey());
  const [entry, setEntry] = useState(emptyEntry());
  const [yesterdayEntry, setYesterdayEntry] = useState(null);
  const [tab, setTab] = useState("study"); // study | urge | history
  const [history, setHistory] = useState([]); // [{key, entry}]
  const [saveState, setSaveState] = useState("idle"); // idle | saving | saved
  const saveTimer = useRef(null);

  // load today + yesterday + recent history on mount
  useEffect(() => {
    (async () => {
      try {
        const todayRes = await window.storage.get(`entry:${dateKey}`);
        if (todayRes) setEntry(JSON.parse(todayRes.value));
      } catch (e) {
        /* no entry yet */
      }
      try {
        const yKey = yesterdayKey(dateKey);
        const yRes = await window.storage.get(`entry:${yKey}`);
        if (yRes) setYesterdayEntry({ key: yKey, ...JSON.parse(yRes.value) });
      } catch (e) {
        setYesterdayEntry(null);
      }
      try {
        const list = await window.storage.list("entry:");
        if (list && list.keys && list.keys.length) {
          const keys = list.keys
            .map((k) => k.replace("entry:", ""))
            .filter((k) => k !== dateKey)
            .sort((a, b) => (a < b ? 1 : -1))
            .slice(0, 14);
          const loaded = [];
          for (const k of keys) {
            try {
              const r = await window.storage.get(`entry:${k}`);
              if (r) loaded.push({ key: k, entry: JSON.parse(r.value) });
            } catch (e) {}
          }
          setHistory(loaded);
        }
      } catch (e) {
        setHistory([]);
      }
      setReady(true);
    })();
  }, [dateKey]);

  // autosave, debounced
  useEffect(() => {
    if (!ready) return;
    setSaveState("saving");
    if (saveTimer.current) clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(async () => {
      try {
        const toSave = { ...entry, loggedAt: new Date().toISOString() };
        const res = await window.storage.set(`entry:${dateKey}`, JSON.stringify(toSave));
        setSaveState(res ? "saved" : "idle");
      } catch (e) {
        setSaveState("idle");
      }
    }, 600);
    return () => clearTimeout(saveTimer.current);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [entry, ready, dateKey]);

  const update = (patch) => setEntry((e) => ({ ...e, ...patch }));

  // streak: consecutive days (including today, if it has any real content) with output logged
  const hasOutput = (e) =>
    !!(e && (e.freeWrite?.trim() || e.oneIdea?.trim() || e.oneAction?.trim()));

  const streak = (() => {
    let count = hasOutput(entry) ? 1 : 0;
    let cursor = dateKey;
    const map = new Map(history.map((h) => [h.key, h.entry]));
    // walk backwards from yesterday
    let k = yesterdayKey(cursor);
    while (map.has(k) && hasOutput(map.get(k))) {
      count += 1;
      k = yesterdayKey(k);
    }
    return count;
  })();

  if (!ready) {
    return (
      <div
        style={{
          minHeight: "100vh",
          background: PAPER,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          fontFamily: "'IBM Plex Mono', monospace",
          color: INK_FAINT,
          fontSize: 13,
        }}
      >
        loading…
      </div>
    );
  }

  return (
    <div
      style={{
        minHeight: "100vh",
        background: PAPER,
        fontFamily: "'IBM Plex Mono', monospace",
      }}
    >
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Newsreader:ital,wght@0,400;0,500;0,600;0,700;1,400;1,500&family=IBM+Plex+Mono:wght@400;500;600&display=swap');
        * { box-sizing: border-box; }
        textarea::placeholder { color: ${INK_FAINT}; opacity: 0.6; }
        button { font-family: inherit; cursor: pointer; }
        .tabbtn { transition: color 0.15s, border-color 0.15s; }
        .chk { transition: background 0.15s, border-color 0.15s; }
      `}</style>

      <div style={{ maxWidth: 640, margin: "0 auto", padding: "22px 16px 60px" }}>
        {/* masthead */}
        <div
          style={{
            display: "flex",
            alignItems: "flex-start",
            justifyContent: "space-between",
            marginBottom: 4,
          }}
        >
          <div>
            <div
              style={{
                fontFamily: "'Newsreader', Georgia, serif",
                fontWeight: 700,
                fontSize: 26,
                color: INK,
                letterSpacing: "-0.015em",
                lineHeight: 1.1,
              }}
            >
              Knowing → Doing
            </div>
            <div
              style={{
                fontSize: 11.5,
                color: INK_FAINT,
                marginTop: 4,
                letterSpacing: "0.02em",
              }}
            >
              {prettyDate(dateKey)}
            </div>
          </div>
          <SeedMark />
        </div>

        <div
          style={{
            fontFamily: "'Newsreader', Georgia, serif",
            fontStyle: "italic",
            fontSize: 14,
            color: INK_FAINT,
            borderBottom: `1px solid ${RULE}`,
            paddingBottom: 16,
            marginBottom: 18,
          }}
        >
          Nothing goes in without something coming out.
        </div>

        {/* streak + save state */}
        <div
          style={{
            display: "flex",
            alignItems: "center",
            justifyContent: "space-between",
            marginBottom: 20,
          }}
        >
          <div style={{ display: "flex", alignItems: "baseline", gap: 8 }}>
            <span
              style={{
                fontFamily: "'Newsreader', Georgia, serif",
                fontWeight: 600,
                fontSize: 22,
                color: streak > 0 ? CLAY : INK_FAINT,
              }}
            >
              {streak}
            </span>
            <span style={{ fontSize: 11.5, color: INK_FAINT }}>
              day{streak === 1 ? "" : "s"} of real output
            </span>
          </div>
          <span style={{ fontSize: 10.5, color: INK_FAINT, letterSpacing: "0.03em" }}>
            {saveState === "saving" ? "saving…" : saveState === "saved" ? "saved" : ""}
          </span>
        </div>

        {/* tabs */}
        <div
          style={{
            display: "flex",
            gap: 4,
            marginBottom: 20,
            borderBottom: `1px solid ${RULE}`,
          }}
        >
          {[
            ["study", "Study & Books"],
            ["urge", "Urge Log"],
            ["history", "Trail"],
          ].map(([key, label]) => (
            <button
              key={key}
              className="tabbtn"
              onClick={() => setTab(key)}
              style={{
                background: "none",
                border: "none",
                borderBottom: tab === key ? `2px solid ${CLAY}` : "2px solid transparent",
                padding: "8px 10px",
                fontSize: 13,
                fontWeight: tab === key ? 600 : 400,
                color: tab === key ? INK : INK_FAINT,
                marginBottom: -1,
              }}
            >
              {label}
            </button>
          ))}
        </div>

        {tab === "study" && (
          <>
            {/* yesterday recall check */}
            <Card style={{ background: MOSS_SOFT, border: `1px solid ${MOSS}` }}>
              <div
                style={{
                  fontFamily: "'Newsreader', Georgia, serif",
                  fontWeight: 600,
                  fontSize: 15,
                  color: "#3E4A34",
                  marginBottom: 8,
                }}
              >
                Before anything new — recall yesterday
              </div>
              {yesterdayEntry && (yesterdayEntry.oneIdea || yesterdayEntry.freeWrite || yesterdayEntry.topic) ? (
                <>
                  <div style={{ fontSize: 12.5, color: "#3E4A34", marginBottom: 10, lineHeight: 1.5 }}>
                    Yesterday's topic was{" "}
                    <strong>{yesterdayEntry.topic || "(untitled)"}</strong>. Close your eyes and try
                    to say it back — out loud — before reading what you wrote. Then check.
                  </div>
                  <button
                    onClick={() => update({ recallDone: !entry.recallDone })}
                    className="chk"
                    style={{
                      display: "flex",
                      alignItems: "center",
                      gap: 8,
                      background: "none",
                      border: "none",
                      padding: 0,
                      marginBottom: entry.recallDone ? 10 : 0,
                    }}
                  >
                    <span
                      style={{
                        width: 18,
                        height: 18,
                        borderRadius: 3,
                        border: `1.5px solid #3E4A34`,
                        background: entry.recallDone ? "#3E4A34" : "transparent",
                        display: "inline-flex",
                        alignItems: "center",
                        justifyContent: "center",
                        color: "#fff",
                        fontSize: 12,
                        flexShrink: 0,
                      }}
                    >
                      {entry.recallDone ? "✓" : ""}
                    </span>
                    <span style={{ fontSize: 13, color: "#3E4A34" }}>
                      I recalled it before checking my notes
                    </span>
                  </button>
                  {entry.recallDone && (
                    <textarea
                      value={entry.recallText}
                      onChange={(e) => update({ recallText: e.target.value })}
                      placeholder="What came back on its own, in your words — even fragments…"
                      rows={2}
                      style={{
                        width: "100%",
                        background: "#FBF8F2",
                        border: `1px solid #A8B594`,
                        borderRadius: 3,
                        padding: "8px 10px",
                        fontFamily: "'IBM Plex Mono', monospace",
                        fontSize: 13,
                        color: INK,
                      }}
                    />
                  )}
                </>
              ) : (
                <div style={{ fontSize: 12.5, color: "#3E4A34", fontStyle: "italic" }}>
                  No entry logged yesterday — start fresh today, and tomorrow this box will hold
                  you to it.
                </div>
              )}
            </Card>

            {/* today's topic */}
            <Card>
              <SectionLabel n="1">What are you studying today?</SectionLabel>
              <Field
                label="Topic"
                placeholder="e.g. Mauryan Administration — APPSC Ch.9, or 'Atomic Habits — Ch.3'"
                value={entry.topic}
                onChange={(v) => update({ topic: v })}
                rows={1}
              />

              <SectionLabel n="2">Read once. Close it. Write from memory.</SectionLabel>
              <Field
                label="Free write — closed book"
                hint="don't peek"
                placeholder="Everything you remember, in your own words. Messy is fine. Stop when it dries up."
                value={entry.freeWrite}
                onChange={(v) => update({ freeWrite: v })}
                rows={6}
              />
              <Field
                label="Gaps you found"
                hint="mark, don't rewrite the whole thing"
                placeholder="Only what you missed when you checked against the source."
                value={entry.gaps}
                onChange={(v) => update({ gaps: v })}
                rows={3}
              />
            </Card>

            <Card>
              <SectionLabel n="3">Compress to three lines</SectionLabel>
              <Field
                label="One idea"
                hint="the single thing this is built around"
                placeholder="If someone remembers nothing else, they remember this."
                value={entry.oneIdea}
                onChange={(v) => update({ oneIdea: v })}
                rows={2}
              />
              <Field
                label="One application"
                hint="to your actual life, not in theory"
                placeholder="Where does this touch something real — your bank work, your prep, your family?"
                value={entry.oneApplication}
                onChange={(v) => update({ oneApplication: v })}
                rows={2}
              />
              <Field
                label="One action"
                hint="this week"
                placeholder="A specific, small, checkable thing you'll actually do."
                value={entry.oneAction}
                onChange={(v) => update({ oneAction: v })}
                rows={2}
              />
            </Card>
          </>
        )}

        {tab === "urge" && (
          <Card>
            <SectionLabel n="✦">In-the-moment log</SectionLabel>
            <div style={{ fontSize: 12.5, color: INK_FAINT, marginBottom: 16, lineHeight: 1.55 }}>
              The plan doesn't hold in the moment — the move does. Log it right after, while it's
              fresh, so you can see what actually works over time instead of guessing.
            </div>
            <button
              onClick={() => update({ urgeLogged: !entry.urgeLogged })}
              className="chk"
              style={{
                display: "flex",
                alignItems: "center",
                gap: 8,
                background: "none",
                border: "none",
                padding: 0,
                marginBottom: 14,
              }}
            >
              <span
                style={{
                  width: 18,
                  height: 18,
                  borderRadius: 3,
                  border: `1.5px solid ${CLAY}`,
                  background: entry.urgeLogged ? CLAY : "transparent",
                  display: "inline-flex",
                  alignItems: "center",
                  justifyContent: "center",
                  color: "#fff",
                  fontSize: 12,
                  flexShrink: 0,
                }}
              >
                {entry.urgeLogged ? "✓" : ""}
              </span>
              <span style={{ fontSize: 13, color: INK }}>An urge showed up today</span>
            </button>

            {entry.urgeLogged && (
              <>
                <Field
                  label="The 90-second move you used"
                  placeholder="What you actually did — walk, cold water, called someone, left the room…"
                  value={entry.urgeMove}
                  onChange={(v) => update({ urgeMove: v })}
                  rows={2}
                />
                <div style={{ marginBottom: 6 }}>
                  <div
                    style={{
                      fontFamily: "'Newsreader', Georgia, serif",
                      fontStyle: "italic",
                      fontSize: 15.5,
                      color: INK,
                      marginBottom: 8,
                    }}
                  >
                    Did it work?
                  </div>
                  <div style={{ display: "flex", gap: 8 }}>
                    {[
                      ["yes", true, MOSS, MOSS_SOFT],
                      ["no", false, CLAY, CLAY_SOFT],
                    ].map(([label, val, color, bg]) => (
                      <button
                        key={label}
                        onClick={() => update({ urgeWorked: val })}
                        style={{
                          flex: 1,
                          padding: "9px 0",
                          borderRadius: 3,
                          border: `1.5px solid ${entry.urgeWorked === val ? color : RULE}`,
                          background: entry.urgeWorked === val ? bg : PAPER_RAISED,
                          color: entry.urgeWorked === val ? color : INK_FAINT,
                          fontSize: 13,
                          fontWeight: entry.urgeWorked === val ? 600 : 400,
                          textTransform: "capitalize",
                        }}
                      >
                        {label}
                      </button>
                    ))}
                  </div>
                </div>
              </>
            )}

            {!entry.urgeLogged && (
              <div style={{ fontSize: 12, color: INK_FAINT, fontStyle: "italic" }}>
                Nothing to log yet today — that's fine. This box is here when you need it, not a
                confession booth.
              </div>
            )}
          </Card>
        )}

        {tab === "history" && (
          <Card>
            <SectionLabel n="↺">Last 14 days</SectionLabel>
            {history.length === 0 && (
              <div style={{ fontSize: 12.5, color: INK_FAINT, fontStyle: "italic" }}>
                Nothing logged yet before today. The trail builds from here.
              </div>
            )}
            {history.map(({ key, entry: e }) => (
              <div
                key={key}
                style={{
                  paddingBottom: 14,
                  marginBottom: 14,
                  borderBottom: `1px solid ${RULE}`,
                }}
              >
                <div
                  style={{
                    display: "flex",
                    justifyContent: "space-between",
                    marginBottom: 6,
                  }}
                >
                  <span
                    style={{
                      fontFamily: "'Newsreader', Georgia, serif",
                      fontWeight: 600,
                      fontSize: 14,
                      color: INK,
                    }}
                  >
                    {prettyDate(key)}
                  </span>
                  <span
                    style={{
                      fontSize: 10.5,
                      color: hasOutput(e) ? MOSS : INK_FAINT,
                      fontWeight: 600,
                    }}
                  >
                    {hasOutput(e) ? "OUTPUT" : "—"}
                  </span>
                </div>
                {e.topic && (
                  <div style={{ fontSize: 12.5, color: INK, marginBottom: 3 }}>
                    <span style={{ color: INK_FAINT }}>topic — </span>
                    {e.topic}
                  </div>
                )}
                {e.oneIdea && (
                  <div style={{ fontSize: 12.5, color: INK, lineHeight: 1.5 }}>
                    <span style={{ color: INK_FAINT }}>idea — </span>
                    {e.oneIdea}
                  </div>
                )}
                {!e.topic && !e.oneIdea && (
                  <div style={{ fontSize: 12, color: INK_FAINT, fontStyle: "italic" }}>
                    empty entry
                  </div>
                )}
              </div>
            ))}
          </Card>
        )}

        <div
          style={{
            textAlign: "center",
            fontSize: 10.5,
            color: INK_FAINT,
            marginTop: 26,
            fontStyle: "italic",
            fontFamily: "'Newsreader', Georgia, serif",
          }}
        >
          Knowing is maybe ten percent. Write it, say it, apply it — that's the rest.
        </div>
      </div>
    </div>
  );
}
